cmake_minimum_required(VERSION 3.20)
project(aibuddy7se_git_wincred LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The Win7 credential helper requires x64 MSVC")
endif()
if(NOT WINCRED_SOURCE)
  set(WINCRED_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/git-credential-wincred.c")
endif()
add_executable(git-credential-wincred "${WINCRED_SOURCE}")
target_compile_definitions(git-credential-wincred PRIVATE WINVER=0x0601 _WIN32_WINNT=0x0601 _CRT_SECURE_NO_WARNINGS)
target_compile_options(git-credential-wincred PRIVATE /W4 /utf-8)
target_link_options(git-credential-wincred PRIVATE /SUBSYSTEM:CONSOLE,6.01)
target_link_libraries(git-credential-wincred PRIVATE Advapi32)
